/|
| Kernregel(.kr) 语言语法示例
| C++ VM 型
| 下列被注释的代码取消注释后可用(注释它只是为了测试时无多余输出)
| 函数名、变量名等均支持 Unicode 字符
| 为了保持语法统一，最好不要使用非默认的可选关键字
| Kr 支持缩进写法(只要缩进大于父级均为正确语法)，但建议保持统一(若父级使用或花括号，子级也必须使用，否则语法错误)；lexikal 不支持缩进式(预处理不允许歧义)
| Kr 为函数作用域，für jedes 遍历会创建新局部作用域，其他循环语句不会。
| 除了内存操作外，任何 Kern 代码都是合法的 Kernregel 代码，执行结果几乎相同但逻辑大不相同
| 已有 Visual Studio Code 语法高亮及 lexikal 等语义高亮拓展(语义高亮可根据用户定义或导入的 lexikal，动态变化高亮，具体操作见下文)
| 语义高亮会调用 Kr 解释器获取 AST、词法 等，以提供错误诊断、标识符详细信息等
|
| 无优化时，百万次迭代，空循环执行耗时约 4 ms，循坏体内有一个递增表达式约 17 ms，相当于 VA 循环约一亿次
| v0.0.19xx 警告: VM 模式可能不如解释器稳定
|

/|
| 执行 Kr 程序时，可在脚本文件名后附加任意数量的参数(以空格分隔)。
| 这些参数会自动注入为两个全局只读常量：
|
|   arga  - 整数，表示附加参数的数量(不包含程序名和脚本文件名)。
|   argw  - 字符串数组，每个元素对应一个参数，索引从 0 开始。
|
| 注意：arga、argw 为常量，不可重新赋值。
|
| 
|   命令行：kr test.kr a b "c d"
|   则 arga = 3，argw = ["a", "b", "c d"]，fname() = "test.kr"
|
| 可与 --opt 同时使用，例如：
|   kr test.kr --opt "{haupt: main}" arg1 arg2
|   此时 main 函数会被调用，而 arga/argw 仍然包含 arg1、arg2。
|
| 用法
|   fkt main() {
|       zeig "程序名: ", argw[0]
|       fa arga > 0 {
|           zeig "第一个参数: ", argw[1]
|       }
|       für i in 0..arga-1 {
|           zeig "argw[", i, "] = ", argw[i]
|       }
|   }
|/

// 缩进写法(同样稳定，只要写法统一，也推荐这种更简洁的写法)
fkt einzug()
    für i in 0..2
        gib 3

    ver
        fa x == 1
            null                // 和 pass 是同一个道理，但实际可以直接留空
        snfa x > 0
            gib 2
        sonst
            dem 0
    den fehl
        zeig fehl

    fa 1
        gib                     // 空返回；因为下一行为空所以无需分号
    // x + 1                    // 这里不会被 gib 识别，因为缩进过浅，此时返回 null
    gib(
        x + 1                   // 括号包裹则无视缩进过浅
    )

// 可变数量参数(`&args`或`...args`)
fkt printf(result, &args) {
    out = result
    fa args == null {
        zeig out
        gib ""
    }
    für i in 0..args.län -1 {
        elem = args[i]
        fa elem == null {
            str = "null"
        } sonst {
            str = elem.alt(".0k")
        }
        out = out.ers("%d", str)
    }
    zeig out
}

// 回响函数；为了防止线程冲突，建议直接用异步函数
kernregel sqrt2(__zweck) {
    x = 1.0
    iterations = 0
    max_iter = 20
    fa __zweck == "compare" {
        max_iter = 5
    }

    indes iterations < max_iter {
        x = (x + 2 / x) / 2
        iterations = iterations + 1
    }
    gib x
}

// 指针操作
/|
| 指针类型：zgr<T> (裸指针) 或 schlau<T> (智能指针)

|   zgr(值)           创建裸指针(复制值到堆)
|   schlau(值)        创建智能指针(复制值到堆，自动管理生命周期)
|   zgrw(指针)        解引用读取
|   zgrs(指针, 值)    解引用写入

        // zeig dies.name + " macht ein Geräusch."

    // 类内声明方法可以省略前缀

    }

// ────────────────────────────────────────────────────────────────
// 语法：{表达式}                      仅求值并插入字符串形式
//       {表达式:alt格式}              调用 alt("alt格式")，交由 alt 处理
//       {表达式:align宽度}            调用 füll(宽度, 方向)
//       {表达式:fill align宽度}       调用 füll(宽度, 方向, fill)
//       {表达式:alt格式 align宽度}    先 alt 后 füll 的链式调用
//
// 对齐字符：
//   <    左对齐(文本靠左，右侧填充)  等价于 füll(N, "r")
//   >    右对齐(文本靠右，左侧填充)  等价于 füll(N, "l")
//   ^    居中(两侧填充)             等价于 füll(N, "c")
//
// fill：可选的单个填充字符，须紧贴 align 字符之前；缺省时用空格。
//       若 fill 位置是一个数字，且它前面也是数字(如 "16>8" 里的 "16")，
//       则该数字属于 alt 段，不算 fill。
//
// 冒号规则：冒号必须位于插值的最外层，且不在字符串、字符、括号内，否则无效。
//           align 与宽度为必需项，且宽度必须是非负整数；否则整段作为 alt 格式。
//
// 反引号插值：反引号内的插值标记为 "&{...}"，普通花括号不受影响。
// ────────────────────────────────────────────────────────────────

// zeig "Hallo {name2}!"               -> "Hallo Welt!"

// —— alt 格式(数值表示) ——
// zeig "{n:.2k}"                      -> "42.00"；直接调用 alt(".2k")，交给 alt 处理
// zeig "{n:16}"                       -> "2A"；进制转换
// zeig "{n:04X}"                      -> "002A"；前导零 + 大写十六进制

// —— 对齐(仅 füll) ——
// zeig "|{n:<8}|"                     -> "|42      |"；左对齐，右侧补空格
// zeig "|{n:>8}|"                     -> "|      42|"；右对齐，左侧补空格
// zeig "|{n:^8}|"                     -> "|   42   |"；居中
// zeig "|{n:*<8}|"                    -> "|42******|"；自定义填充符 '*'
// zeig "|{n:-^8}|"                    -> "|---42---|"；填充符 '-'
// zeig "|{n:0>4}|"                    -> "|0042|"；数字补零，此处 '0' 是填充符

// —— alt 与对齐组合(先 alt 再 füll) ——
// zeig "|{n:16>8}|"                   -> "|      2A|"；先转十六进制，再右对齐
// zeig "|{n:16<8}|"                   -> "|2A      |"；先转十六进制，再左对齐
// zeig "|{n:16^8}|"                   -> "|   2A   |"；先转十六进制，再居中
// zeig "|{n:16*<8}|"                  -> "|2A******|"；alt="16"，fill='*'，左对齐
// zeig "|{n:.2k>10}|"                 -> "|     42.00|"；先保留两位小数，再右对齐

// —— alt 与对齐的边界情况 ——
// zeig "{n:.2k^}"                     -> "42.00"；向上取整。因 '^' 后无数字，仍属于 alt 段
// zeig "{n:04X}"                      -> "002A"；无 align 字符，整段走 alt
// zeig "{n:2>4}"                      -> "**42"；'2' 前无数字，故视为填充符，等价于 n.füll(4, "l", "2")
// zeig "{n:16>8}"                     -> "      2A"；'6' 前有数字，故 '16' 属于 alt 段

// —— 显式调用(格式串本身以数字结尾且想用数字做填充符时) ——
// zeig "|{n.füll(8, "l", "2")}|"      -> "|22222242|"；绕开歧义，显式传参

// 启动服务器 (starteServer / ac_starteServer)
// ────────────────────────────────────────────────────────────────────────────
// 处理函数接收一个请求对象 anfrage：
//   anfrage.methode  - HTTP 方法 ("GET" / "POST" / "HEAD" / "PUT" / ...)
//   anfrage.pfad     - 完整请求路径(含查询串)
//   anfrage.kopf     - 请求头对象 (键为头字段名)
//   anfrage.inhalt   - 请求体字符串(未解析)
//
// 处理函数必须返回一个响应对象：
//   { zustand: 状态码, kopf: { ...响应头... }, inhalt: 响应体 }
//
// 内置响应构造器(推荐使用，自动设置 Content-Type)：
//   html(s [, status])        - HTML 响应(Content-Type: text/html; charset=utf-8)
//   json(v [, status])        - JSON 响应(Content-Type: application/json; charset=utf-8)
//   text(s [, status])        - 纯文本响应(Content-Type: text/plain; charset=utf-8)
//   umleiten(url [, status])  - 重定向(Location 头；默认 302，可传 301 / 303 / 307 / 308)
//
// 请求解析辅助：
//   abfrage(url)              - 解析 URL 查询串 → { key: value }
//   formular(body)            - 解析 application/x-www-form-urlencoded → { key: value }
//   esc(s)                    - HTML 实体转义(& < > " ')
//   kodierURL(s)              - URL 百分号编码
//   dekodierURL(s)            - URL 百分号解码
//
// 流式响应 (SSE)：
//   SSE()                     - 创建事件流对象
//   sende(stream, data)       - 发送消息(自动加 "data:" 前缀)
//   sendeSSE(stream, ev, data)- 发送命名事件(ev 为空则等同 sende)
//   schliesse(stream)         - 关闭流(客户端 EventSource 会收到 EOF)
//
// 注意：处理函数不应阻塞。SSE 生产者必须放在 ac fkt 中异步执行，
//       否则 MHD 线程无法返回响应头，浏览器会一直转圈。

// ---------- 示例 1: 基础路由(GET / 响应构造器 / 查询参数) ----------
fkt handler(anfrage) {
    url    = anfrage.pfad
    params = abfrage(url)               // { key: value, ... }
    qm     = url.stelleVon("?")
    pfad   = qm == -1 ? url : url.ab(0, qm)

    fa pfad == "/" {
        titel = params.titel ?? "Welt"
        gib html(`
            <!DOCTYPE html>
            <html lang="de">
            <head>
                <meta charset="utf-8">
                <title>Kernregel</title>
                <style>
                    body { font-family: sans-serif; margin: 2em; }
                    h1   { color: #2a6; }
                    code { background: #eee; padding: 2px 4px; border-radius: 3px; }
                </style>
            </head>
            <body>
                <h1>Hallo &{esc(titel)}!</h1>
                <p>Pfad: <code>&{esc(pfad)}</code></p>
                <p>Suche: <code>&{esc(params.suche ?? "—")}</code></p>
                <ul>
                    <li><a href="/daten?x=1&y=2">/daten?x=1&amp;y=2</a> (JSON)</li>
                    <li><a href="/text?name=Max">/text?name=Max</a> (Klartext)</li>
                    <li><a href="/alt">/alt</a> (Weiterleitung → /)</li>
                </ul>
            </body>
            </html>
        `)
    }
    snfa pfad == "/daten" {
        gib json({
            pfad: pfad,
            parameter: params,
            zeit: zeit(),
            gefunden: wahr
        })
    }
    snfa pfad == "/text" {
        gib text("Hallo " ## (params.name ?? "Welt") ## "!\n")
    }
    snfa pfad == "/alt" {
        gib umleiten("/", 302)
    }
    sonst {
        gib html(`
            <h1>404 - Nicht gefunden</h1>
            <p>Angefragt: <code>&{esc(url)}</code></p>
        `, 404)
    }
}
starteServer(8080, handler)

puffer: dq 0, 0, 0, 0            ; 32 字节缓冲区，供内存寻址演示使用
zahl:   dq 42                    ; 64 位整数
hexwer: db 0xFF, 0x0A            ; 两个字节：255 和 10
arr:    dd 1, 2, 3, 4            ; 四个 32 位整数
wort:   dw 0x1234                ; 16 位整数(0x1234)

; ---------- 程序入口(固定 _start) ----------
_start:
    ; ---------- 传送 (der) ----------
    der r1, puffer          ; r1 = puffer 的运行时地址
    der r2, 100             ; 写入值
    der [r1], r2            ; *(int64*)r1 = 100
    der r3, [r1]            ; r3 = *(int64*)r1 = 100
    ; 或者用偏移
    der r2, 200
    der [r1+8], r2
    der r4, [r1+8]          ; r4 = 200

    ; ---------- 算术运算 ----------
    add r1, 3               ; r1 += 3
    sub r4, r2              ; r4 -= r2
    ink r1                  ; r1++
    dek r4                  ; r4--

    ; ---------- 乘法(单操作数) ----------
    mul r4                  ; r1 = r1 * r4(被乘数隐含为 r1，结果在 r1)
    ; 注：单操作数 mul 使用无符号乘法(mul r/m64)，破坏 r1、r3

    ; ---------- 乘法(多操作数) ----------
    mul r4, r2              ; r4 = r4 * r2          (2 操作数：dest = dest * src)
    mul r13, r1, 3          ; r13 = r1 * 3          (3 操作数：dest = src1 * src2)
    mul r14, r1, r2, r4     ; r14 = r1 * r2 * r4    (4+ 操作数：连乘)
    mul r7, r1, r2, 5, r4   ; r7 = r1 * r2 * 5 * r4
    ; 注：多操作数版本使用有符号乘法(imul)，仅修改目标寄存器，不破坏 r1/r3；
    ;     第一个操作数(目标)必须是寄存器；操作数数量不限
    der r3, 0               ; r3 = RDX = 0

    ; ---------- 除法(单操作数) ----------
    div r2                  ; r1 = r1 / r2, r3 = r1 % r2
    ; 注：单操作数 div 使用无符号除法(div r/m64)，被除数隐含为 r1:r3 组成的 128 位数，
    ;     结果商在 r1，余数在 r3

    ; ---------- 除法(多操作数) ----------
    div r4, r2                          ; r4 = r4 / r2          (2 操作数：商存入 r4，余数丢弃)
    div r15, r1, r2                     ; r15 = r1 / r2         (3 操作数：商存入 r15，余数丢弃)
    div r16, r7, r1, r2                 ; r16 = r1 / r2         (4 操作数：商存入 r16)
                                        ; r7 = r1 % r2          (余数存入 r7)
    ; 注：多操作数版本使用无符号除法(div r/m64)，除数通过栈临时保存，
    ;     不会破坏其它寄存器(商/余数按需写入指定寄存器)；
    ;     第一个操作数(商)必须是寄存器；
    ;     4 操作数形式的第二个操作数(余数)也必须是寄存器；
    ;     除数可以是寄存器、立即数(<= 32 位)或内存操作数

    ; ---------- 位运算 ----------
    und r1, 0xFF            ; r1 &= 0xFF
    od  r4, r2              ; r4 |= r2
    xod r1, 5               ; r1 ^= 5
    nich r1                 ; r1 = ~r1(按位取反)

    ; ---------- 比较(结果存于 r3)----------
    vgl r1, 10              ; 比较 r1 与 10
                            ; r3 = 0(相等), 1(r1 > 10), -1(r1 < 10)

    ; ---------- 条件跳转 ----------
    spr marke1              ; 无条件跳转到 marke1
    ; 注意：以下跳转指令都依赖 r3 的值，由于上面 spr 已经跳转，这些指令不会执行
    sn  marke2              ; 若 r3 == 0 则跳转
    snn marke3              ; 若 r3 != 0 则跳转
    sg  marke4              ; 若 r3 == 1(大于)则跳转
    sk  marke5              ; 若 r3 == -1(小于)则跳转
    sge marke6              ; 若 r3 >= 0 则跳转
    ske marke7              ; 若 r3 <= 0 则跳转

marke1:
    ; ---------- 子程序调用 ----------
    ruf unterprog           ; 调用子程序 unterprog(返回地址压栈)
    ; 注意：这里调用后，子程序执行完毕会通过 gib 返回此处，继续执行下一条指令
    spr ende_s              ; 跳转到程序结束标签(直接结束，不再执行后面的代码)

unterprog:
    ; ---------- 堆栈操作 ----------
    drk r1                  ; 将 r1 的值压栈
    hol r4                  ; 弹栈至 r4(此时 r4 = 原先的 r1)
    gib                     ; 返回调用点(即 ruf 后面的那条指令)

    ; ---------- 软件中断----------
    ; 注意：该部分在 unterprog 之后，但由于 gib 已返回，不会被执行。
    ; 若要使用，应将其放在主流程中。
    der r1, 1               ; 功能 1 = 退出
    unt 0x80

    ; ---------- 打印字符串----------
    zeig msg                ; 输出 msg 指向的字符串

    ; ---------- 程序退出 ----------
    ende                    ; 结束程序

; ---------- 仅作跳转目标的标签 ----------
marke2:
marke3:
marke4:
marke5:
marke6:
marke7:
    ; 这些标签是空的，仅用于演示跳转目标，实际应用中可放置代码

    ; ---------- 计数循环示例 ----------
    der r1, 0               ; 初始化计数器
schleife:
    vgl r1, 10              ; 比较 r1 和 10
    sge ende_s              ; 若 r1 >= 10 则跳出循环
    ink r1                  ; r1++
    spr schleife            ; 跳回循环开始
ende_s:
    ende                    ; 程序结束(无论从哪个路径到达此处都退出)
    ; --------- 其他指令 ---------
    ugib                    ; 中断返回(x86-64 下生成 IRETQ)
    in  r1, 0x60            ; 从 I/O 端口 0x60 读入到 r1(RAX)
    erg 0x60, r1            ; 将 r1(RAX) 写出到 I/O 端口 0x60

; ============================================================================
; 补充说明(逻辑详解)：
; 1. 程序入口固定为 _start，执行顺序为从上到下，遇到跳转则改变 PC。
; 2. 子程序(如 unterprog)必须通过 ruf 调用，内部以 gib 返回。
;    主程序(_start 或任何非子程序段)不得使用 gib，应使用 ende 结束。
; 3. 数据标签(如 msg)只能用于数据操作(如 zeig 或 der 取地址)，
;    不能作为跳转目标(否则解释器会报“未知标签”)。
; 4. 条件跳转(sn, sgr 等)依赖 r3 寄存器的比较结果(由 vgl 设置)。
; 5. 内存寻址格式：[reg+disp] 或 [reg-disp]，disp 为 64 位有符号整数。
; 6. 数值支持十进制和十六进制(0x...)，均为 64 位有符号。
; 7. 解释器模式下，unt 0x80 支持功能 1(退出)和 4(写控制台)；
;    编译模式(-c exe)下，zeig 和 ende 转为 WinAPI 调用(仅 Windows)。
; ============================================================================
